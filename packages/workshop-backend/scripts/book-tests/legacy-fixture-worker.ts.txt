// Local diagnostic entrypoint for the approved isolated legacy-data generation only.
import actual from './server';
import { OverseerDurableObject as ActualOverseer, UserDurableObject as ActualUser } from './server';
import { DurableObject } from 'cloudflare:workers';
import * as Y from 'yjs';
export * from './server';
export class UserDurableObject extends ActualUser {
  fixtureKv() { return [...this.ctx.storage.kv.list()]; }
}
export class BookInspectionFacet extends DurableObject {
  inspect() { return Object.fromEntries(['book_files','progress','messages','settings'].map(name => [name,[...this.ctx.storage.sql.exec('SELECT * FROM '+name)]])); }
}
export class OverseerDurableObject extends ActualOverseer {
  async fixtureState(ownerId: string, gadgetId: number, prepare: boolean) {
    if (this.ctx.storage.kv.get('ownerId') !== ownerId) throw new Error('Fixture owner mismatch');
    const impl=(this as any).impl;
    if (prepare) {
      using facet=await impl.getGadgetFacet(gadgetId);
      await facet.setChapterComplete('legacy',true);
      await facet.askTutor({message:'Explain the preserved chapter in one sentence.',chapterId:'legacy',chapterText:'# Legacy chapter\n\nA preserved manuscript with $x^2$.'});
      return null;
    }
    return {kv:[...this.ctx.storage.kv.list()], update:Y.encodeStateAsUpdateV2(impl.buildYDoc('current').ydoc),root:impl.gadgetRootName(gadgetId)};
  }
  async fixtureTables(ownerId: string,gadgetId: number) {
    if (this.ctx.storage.kv.get('ownerId') !== ownerId) throw new Error('Fixture owner mismatch');
    const name=(this as any).impl.gadgetFacetName(gadgetId);
    this.ctx.facets.abort(name,new Error('Stopped isolated old book for actual SQL capture'));
    return this.ctx.facets.get(name,()=>({class:(this.ctx.exports as any).BookInspectionFacet})).inspect();
  }
}
function encode(value: any): any {
  if(value instanceof Date)return {$date:value.toISOString()};
  if(value instanceof Uint8Array)return {$bytes:value.toBase64()};
  if(Array.isArray(value))return value.map(encode);
  if(value && typeof value==='object')return Object.fromEntries(Object.entries(value).map(([k,v])=>[k,encode(v)]));
  return value;
}
export default {
  async fetch(req: Request,env: Cloudflare.Env,ctx: ExecutionContext) {
    if(new URL(req.url).pathname!=='/__isolated_book_fixture')return actual.fetch(req,env,ctx);
    const args=await req.json() as any;
    const user=(ctx.exports as any).UserDurableObject.getByName(args.username);
    const [tokenUsername,sessionToken]=args.token.split(':');
    if(tokenUsername!==args.username)throw new Error('Fixture session owner mismatch');
    await user.authenticate(sessionToken);
    const ownerId=user.id.toString();
    if(args.operation==='owner')return Response.json(encode({ownerId,kv:await user.fixtureKv()}));
    const ns=(ctx.exports as any).OverseerDurableObject;
    const workspace=ns.get(ns.idFromString(args.workspaceId));
    if(args.operation==='book')return Response.json(encode(await workspace.getBookMcpWorkspace(ownerId)));
    if(args.operation==='put')return Response.json(encode(await workspace.putBookMcpFiles(ownerId,args.files,args.gadgetId)));
    const value=args.operation==='tables' ? await workspace.fixtureTables(ownerId,args.gadgetId) : await workspace.fixtureState(ownerId,args.gadgetId,args.operation==='prepare');
    return Response.json(encode(value));
  }
};
